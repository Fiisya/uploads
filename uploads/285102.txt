const BASE_URL = 'https://api.vidssave.com/api/contentsite_api'
const AUTH = '20250901majwlqo'
const DOMAIN = 'id.vidssave.com'

/**
 * POST helper - application/x-www-form-urlencoded
 */
async function post(path, params) {
  const body = new URLSearchParams({ auth: AUTH, domain: DOMAIN, ...params })
  const res = await fetch(`${BASE_URL}/${path}`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: body.toString()
  })
  if (!res.ok) throw new Error(`HTTP ${res.status} on ${path}`)
  return res.json()
}

/**
 * Internal: parse one origin (cache or source)
 */
async function _parse(url, origin) {
  const data = await post('media/parse', { origin, link: url })
  if (data.status === 1) return data.data
  if (data.status === -1) throw new Error('captcha failed')
  return null
}

/**
 * Format resources array → { video, audio }
 */
function _formatResources(resources) {
  const video = []
  const audio = []

  for (const r of resources) {
    const item = {
      quality: r.quality,
      format: r.format || 'MP4',
      size: r.size || 0,
      // download_url: siap dipakai kalau ada, null kalau butuh resolveDownload()
      download_url: r.download_url || null,
      resource_id: r.resource_id,
      // download_mode referensi: "" | "direct" | "check_download" | "popup"
      download_mode: r.download_mode || '',
      has_audio: r.has_audio ?? null,
      isFast: r.isFast ?? (r.download_mode === 'direct')
    }
    if (r.type === 'video') video.push(item)
    else if (r.type === 'audio') audio.push(item)
  }

  // Dedup by quality (ambil yang punya download_url, prefer direct mode)
  const dedup = (arr) => {
    const map = new Map()
    for (const item of arr) {
      const key = item.quality
      const existing = map.get(key)
      if (!existing || (item.download_url && !existing.download_url)) {
        map.set(key, item)
      }
    }
    return [...map.values()]
  }

  // Sort ascending quality number
  const qualNum = (q) => {
    const n = parseInt(q)
    if (!isNaN(n)) return n
    // audio: LOW=0, 48KBPS=48, 128KBPS=128, etc.
    const kbps = q.match(/(\d+)KBPS/)
    if (kbps) return parseInt(kbps[1])
    return 0
  }
  const sort = (arr) => arr.sort((a, b) => qualNum(a.quality) - qualNum(b.quality))

  return {
    video: sort(dedup(video)),
    audio: sort(dedup(audio))
  }
}

/**
 * Parse video info dari URL.
 * Strategy: coba origin=cache dulu (cepat, direct URLs), fallback ke origin=source.
 *
 * @param {string} url - URL video/audio
 * @param {'cache'|'source'|'auto'} origin - default 'auto' (cache then source)
 * @returns {{ title, thumbnail, duration, video, audio }}
 *   video/audio = array of { quality, format, size, download_url, resource_id, download_mode, has_audio, isFast }
 */
export async function parse(url, origin = 'auto') {
  let raw = null

  if (origin === 'cache' || origin === 'auto') {
    raw = await _parse(url, 'cache')
  }

  if ((!raw || !raw.resources?.length) && (origin === 'source' || origin === 'auto')) {
    raw = await _parse(url, 'source')
  }

  if (!raw) throw new Error('parse failed: no data from both origins')

  const { title, thumbnail, duration, resources } = raw
  const { video, audio } = _formatResources(resources)

  return { title, thumbnail, duration, video, audio }
}

/**
 * Resolve download URL untuk resource yang download_url=null.
 * Submits task → polls SSE → returns download_link.
 *
 * @param {string} resource_id
 * @param {number} timeoutMs - max wait ms (default 60s)
 * @returns {string} download_link (direct URL)
 */
export async function resolveDownload(resource_id, timeoutMs = 60_000) {
  // Step 1: submit task
  const taskRes = await post('media/download', { resource_id })
  if (taskRes.status !== 1) throw new Error(`submit task failed: ${taskRes.msg}`)
  const task_id = taskRes.data.task_id

  // Step 2: stream SSE
  const body = new URLSearchParams({ auth: AUTH, domain: DOMAIN, task_id })
  const res = await fetch(`${BASE_URL}/media/download_query`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body: body.toString(),
    signal: AbortSignal.timeout(timeoutMs)
  })

  const text = await res.text()

  // Parse SSE: "event: X\ndata: {...}\n\n"
  for (const block of text.split('\n\n')) {
    const evtLine = block.match(/^event:\s*(.+)/m)
    const dataLine = block.match(/^data:\s*(.+)/m)
    if (!dataLine) continue
    const d = JSON.parse(dataLine[1])
    const evt = evtLine?.[1]?.trim()
    if (evt === 'success' || d.status === 'success') return d.download_link
    if (evt === 'failed' || d.status === 'failed') throw new Error('task processing failed on server')
  }

  throw new Error('SSE stream ended without success event')
}
